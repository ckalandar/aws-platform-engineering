ECS Tasks

# Set up environment
 export AWS_ENDPOINT_URL=http://localhost:4566

 # Register task definition and create cluster
 aws ecs register-task-definition \

    --family web-task --network-mode bridge \

    --container-definitions '[{"name":"web","image":"nginx:alpine","portMappings":[{"containerPort":80,"hostPort":8081}],"memory":128,"cpu":64}]'


aws ecs create-cluster --cluster-name dev-cluster

 # Run a task (Floci launches a real Docker container)
 aws ecs run-task \

    --cluster dev-cluster \

    --task-definition web-task \

    --count 1

Athena + S3

# Upload data to S3
 aws s3 mb s3://analytics-bucket

aws s3 cp orders.json s3://analytics-bucket/orders/

 # Register Glue table and run SQL query
 aws glue create-database --database-input '{"Name":"shop"}'


QUERY_ID=$(aws athena start-query-execution \

    --query-string "SELECT customer, SUM(amount) FROM shop.orders GROUP BY customer" \

    --result-configuration '{"OutputLocation":"s3://analytics-bucket/results/"}' \

  --query 'QueryExecutionId' --output text)


aws athena get-query-results --query-execution-id $QUERY_ID
